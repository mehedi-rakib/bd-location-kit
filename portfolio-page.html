Installation

Quick Example

const { getAreasByDistrict } = require('bd-location-kit'); // Get all areas in Feni district const feniAreas = getAreasByDistrict(2); console.log(feniAreas); /* Output: [ { id: 18, name: 'Chhagalnaiya', bn_name: 'ছাগলনাইয়া', district_id: 2 }, { id: 19, name: 'Feni Sadar', bn_name: 'ফেনী সদর', district_id: 2 }, { id: 20, name: 'Sonagazi', bn_name: 'সোনাগাজী', district_id: 2 }, ... ] */
